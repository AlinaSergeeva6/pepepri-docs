'use strict';

/* About this script
 * What is does: compares installed addons of 2 environments and outputs differences
 * How to use:
 * 1. Export UDTs list: User Defined Tables -> burger menu -> Export. Remove line with 'ADDON_CPI_SIDE_DATA' UDT.
 * 2. Convert it to JSON and paste it into 'tablesToCreate' variable.
 * 3. In destination env open https://app.pepperi.com/settings_block/354c5123-a7d0-4f52-8fce-3cf1ebc95314/user_defined_tables?view=config_mapdata&uri=grid%2Fmapdatametadata
 * 4. Open console and select UDT settings iframe as execution context (myFrame (app/) settings.pepperi.com)
 * 5. Paste and execute this code
 */

// config start
const tablesToCreate = [];
const runInParallel = false; // set to true if you want to post data in parallel. NOT recommended to use, as it is legacy API
const chunkSize = 5; // specify size of each sub array for parallel run
// config end

const typesMapping = {
    'Sales Rep External ID': 23,
    'Account External ID': 35,
    Any: 0,
    'Catalog Name': 54,
};

const startTime = new Date().getTime();
const postArr = [];

async function getJson(data) {
    console.log(`Starting posting tables ${runInParallel ? 'in parallel' : 'one by one'}..`);
    if (runInParallel) {
        const chunkedArray = chunkify(data, chunkSize);
        await Promise.all(
            chunkedArray.map(async (chunk) => {
                for (let entity of chunk) {
                    const resp = await postEntity(entity);
                    postArr.push(resp);
                }
            })
        );
    } else {
        for (let entity of data) {
            const resp = await postEntity(entity);
            postArr.push(resp);
        }
    }
    console.log(`posting ${tablesToCreate.length} tables took ${new Date().getTime() - startTime}ms`);
    console.log('POST result:');
    console.log(JSON.stringify(postArr));
}

async function postEntity(table) {
    try {
        const memoryType = table['Memory Management'];
        const response = await fetch('https://settings.pepperi.com/Views/Global/AjaxGlobalMethods.aspx/SaveStudioGridModel', {
            headers: {
                accept: 'application/json, text/javascript, */*; q=0.01',
                'accept-language': 'en-GB,en-US;q=0.9,en;q=0.8',
                'content-type': 'application/json; charset=UTF-8',
                'sec-fetch-dest': 'empty',
                'sec-fetch-mode': 'cors',
                'sec-fetch-site': 'same-origin',
                'x-requested-with': 'XMLHttpRequest',
            },
            referrer: 'https://settings.pepperi.com/app/',
            referrerPolicy: 'strict-origin-when-cross-origin',
            body: `{"model":{"ExternalID":"${table['API Name']}","Key1Type":${typesMapping[table['Main Key']]},"Key2Type":${
                typesMapping[table['Secondary Key']]
            },"Dormant":false,"Volatile":false,"Configuration":"{\\"Dormant\\":${memoryType == 'Dormant' ? 'true' : 'false'},\\"Volatile\\":${
                memoryType == 'Volatile' ? 'true' : 'false'
            }}"},"listOptions":{"Page":1,"Table":"mapdatametadata","EntityID":0}}`,
            method: 'POST',
            mode: 'cors',
            credentials: 'include',
        });
        const data = await response.json();
        return data;
    } catch (e) {
        console.log(e);
    }
}
function chunkify(arr, numOfChunks) {
    let res = [];
    let i = 0;
    let chunkSize = Math.ceil(arr.length / numOfChunks);
    while (i < arr.length) {
        res.push(arr.slice(i, i + chunkSize));
        i += chunkSize;
    }
    return res;
}

getJson(tablesToCreate);
