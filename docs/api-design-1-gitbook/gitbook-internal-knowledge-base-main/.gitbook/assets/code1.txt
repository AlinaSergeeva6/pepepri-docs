  var parentUUID = workflowObject.TSAParentUUID;

        fields:['ItemUUID','ItemExternalID','UnitsQuantity','UnitPriceAfterDiscount'],
        filter:{
            Operation : "AND",
            LeftNode: {
                        ApiName: "TSADepartmentID",
                        Operation: "IsEqual",
                        Values: [parentItemExternalID.toString()]},
            RightNode:{
                        ApiName: "Transaction.UUID",
                        Operation: "IsEqual",
                        Values: [parentUUID.toString()]}
        },
        responseCallback:'callback1'
});
    function callback1(response){
            var objs = response.objects;
            var objtoupdate = [];
            for (i in objs){
          objtoupdate.push({
            item:{UUID:objs[i].ItemUUID},
                        UnitsQuantity: objs[i].UnitsQuantity,
                        UnitPriceAfterDiscount: objs[i].UnitPriceAfterDiscount
          })
        }
            pepperi.app.transactionScopeItems.update({
                                    transaction:{UUID: workflowObject.UUID},
                                    objects: objtoupdate,
            responseCallback: 'callback2'
                        });
    }
