  <script>
var parentUUID = workflowObject.TSAParentUUID;
    var parentItemExternalID = workflowObject.TSAParentItemExternalID;
    pepperi.api.transactionLines.search({
        fields:['Transaction.UUID','UUID'],
        filter:{
        	Operation : "AND", 
          	LeftNode: {
              	ApiName: "TSADepartmentID",
          		Operation: "IsEqual", 
          		Values: [parentItemExternalID]},
          	RightNode:{
              	ApiName: "Transaction.UUID",
          		Operation: "IsEqual", 
          		Values: [parentUUID]} 
        },
        responseCallback:'callback1'
      })


//Remove them:
    function callback1(response){
      if (response.count==0) onSaveAndClose();
      	var obj = response.objects;
      	var to_update = []
        for (i in obj){
          to_update.push({
            UUID: obj[i].UUID
          })
        }
    pepperi.app.transactions.removeLines({
          	transaction:{UUID:obj[0]['Transaction.UUID']},
 			lines: to_update,
          	responseCallback: 'callback2'
        })
    }


//Then we should find and delete separately transaction line with department information:
    function callback2(response){
       pepperi.api.transactionLines.search({
        	fields:['Transaction.UUID','UUID'],
        	filter:{
        	Operation : "AND", 
          	LeftNode: {
              	ApiName: "ItemCode",
          		Operation: "IsEqual", 
          		Values: [parentItemExternalID]},
          	RightNode:{
              	ApiName: "Transaction.UUID",
          		Operation: "IsEqual", 
          		Values: [parentUUID]} 
        },
        responseCallback:'callback3'
      })
    }
    
    function callback3(response){
       pepperi.app.transactions.removeLines({
          	transaction:{UUID:response.objects[0]['Transaction.UUID']},
 			lines: [{UUID:response.objects[0]['UUID']}],
          	responseCallback: 'callback4'
        })
    }
    
    function callback4(response){
       onSaveAndClose()
      }


  </script>
