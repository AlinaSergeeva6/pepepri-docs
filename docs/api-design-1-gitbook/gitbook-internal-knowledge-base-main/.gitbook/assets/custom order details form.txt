    <script>
      //SCRIPTS FOR LOGIC
      /**
        * * Description of the function
      */
      const cSOrderSubmittedBy = document.getElementById('mantadoryVal');
      const cSHoldCode = document.getElementById('csholdcode');
      const s1ReasonCode = document.getElementById('s1ReasoneCode');
      const editableNAccPass = document.getElementById('accPass');
      const editableNAuthBuyers = document.getElementById('authBuyers');
      const editableNShipInst = document.getElementById('shipInst');
      const editableNOthrComments = document.getElementById('othrComments');
      
      var falseElement = `<span class="falseElement elembox"> ❌ </span>`;
      var trueElement = `<span class="trueElement elembox"> ✅ </span>`;
      var textLable = `<span class="elembox">Send Order As Replacement (S1)</span>`;


      var checkVal = false; //for boolean
      
      var globalData={
        cfContext: [],
        fields:{},
        needFields: [
            "TSACSOrderSubmittedBy", 
            "TSACSHoldCode", 
            "TSASendOrderAsS1ToJDE",
            "TSAS1ReasonCode",
            "TSAEditableNotesAccountPassword",
            "TSAEditableNotesAuthorizedBuyers",
            "TSAEditableNotesShippingInstructions",
            "TSAEditableNotesOtherComments"
        ],
      };
      pepperi.app.getContext({responseCallback: "getContAndCheck"});

      /**
        ! * Description of the function getContAndCheck
          * Just Callback to get fields from transaction API
      */
      function getContAndCheck(data){
        globalData.cfContext = data;
        
        //Get Fields from transaction
        pepperi.api.transactions.get({
          key: { UUID: globalData.cfContext.transaction.uuid },
          fields: [
            ...globalData.needFields
          ],
          responseCallback: "fieldsInsertToHtml"
        });
      };
      
      /**
        ! * Description of the function fieldsInsertToHtml
          * If fields exist in transaction we just put that to html
      */
      function fieldsInsertToHtml(fieldsData){
        globalData.fields = fieldsData.object;
        
        //Check if fields empty
        if(fieldsData.success){
          var isAllValuesNull = Object.values(globalData.fields).every(val => val === null);
        };
        
        //add values to input if no empty
        if (isAllValuesNull!=true) {

            const {
                TSACSOrderSubmittedBy,
                TSACSHoldCode,
                TSASendOrderAsS1ToJDE,
                TSAS1ReasonCode,
                TSAEditableNotesAccountPassword,
                TSAEditableNotesAuthorizedBuyers,
                TSAEditableNotesShippingInstructions,
                TSAEditableNotesOtherComments,
            } = globalData.fields;

            cSOrderSubmittedBy.value = TSACSOrderSubmittedBy;
            cSHoldCode.innerHTML = TSACSHoldCode;
            addBoolFromField(TSASendOrderAsS1ToJDE);
            s1ReasonCode.innerHTML = TSAS1ReasonCode;
            editableNAccPass.value = TSAEditableNotesAccountPassword;
            editableNAuthBuyers.value = TSAEditableNotesAuthorizedBuyers;
            editableNShipInst.value = TSAEditableNotesShippingInstructions;
            editableNOthrComments.value = TSAEditableNotesOtherComments;
            
        } else { addBoolFromField(false) };
      };
      
      /**
        ! * Description of the function updateDetails
          * Action onClick for button where we update fields to transaction
          * Check mandatory input; if empty don't close and don't update that
      */
      function updateDetails(){
        if(cSOrderSubmittedBy.value == ''){
          cSOrderSubmittedBy.classList.add('warn');
        } else {
          try{
            cSOrderSubmittedBy.classList.remove('warn');
            pepperi.api.transactions.update({
              objects: [{
                UUID: globalData.cfContext.transaction.uuid,
                TSACSOrderSubmittedBy: cSOrderSubmittedBy.value,
                TSACSHoldCode: cSHoldCode.outerText,
                TSASendOrderAsS1ToJDE: !checkVal,
                TSAS1ReasonCode: s1ReasonCode.outerText,
                TSAEditableNotesAccountPassword: editableNAccPass.value,
                TSAEditableNotesAuthorizedBuyers: editableNAuthBuyers.value,
                TSAEditableNotesShippingInstructions: editableNShipInst.value, 
                TSAEditableNotesOtherComments: editableNOthrComments.value 
              }]
            });
            console.log('Updated');
            window.pepperiApp.onClose();
          } catch(e) {
            console.log('not updated(');
          }
        }
      };
    //Logic Scripts End

    //Style Scripts SCR2 Start
    
      //SCRIPTS FOR DOM
      //DropDown Menu
      let index = 1;

      const on = (listener, query, fn) => {
        document.querySelectorAll(query).forEach((item) => {
          item.addEventListener(listener, (el) => {
            fn(el);
          });
        });
      };

      on("click", ".selectBtn", (item) => {
        const next = item.target.nextElementSibling;
        next.classList.toggle("toggle");
        next.style.zIndex = index++;
      });
      on("click", ".option", (item) => {
        item.target.parentElement.classList.remove("toggle");

        const parent = item.target.closest(".select").children[0];
        parent.setAttribute("data-type", item.target.getAttribute("data-type"));
        parent.innerText = item.target.innerText;
      });

      document.getElementById('bool_container').addEventListener('click', function(){   
        if(checkVal==false){
          document.getElementById('bool_container').innerHTML =`
            ${falseElement}
            ${textLable}
          `;
          checkVal = true;
        } else {
          document.getElementById('bool_container').innerHTML =`
            ${trueElement}
            ${textLable}
          `;
          checkVal = false;
        };
      });
      
      function addBoolFromField(boolVal){
        if(boolVal){
          document.getElementById('bool_container').innerHTML =`
            ${trueElement}
            ${textLable}
          `;
          checkVal = false;
        } else {
          document.getElementById('bool_container').innerHTML =`
            ${falseElement}
            ${textLable}
          `;
          checkVal = true;
        };
      };
    </script>
