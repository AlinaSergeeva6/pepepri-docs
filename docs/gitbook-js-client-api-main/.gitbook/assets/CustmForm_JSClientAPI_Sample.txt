  function callback_context_working()
  {
    	//idea:	callback method belongs to a global object defined on window
    
    	//BL: create dummy window.key1
    	window.key1 					= "method pailed value";
    	window.callback_context_object 	=
    	{
      		key1: "method passed value",
          
        	callback_method:  function()
        	{
          		alert ("in callback method. this.key1=" + this.key1 );	//this will be the this of the object
        	}
    	};
    
    	//BL: call client api
    	var mainKey = "1";
     	var secondaryKey = "";
     	pepperi.api.userDefinedTables.get({
    		table:"integration_ajax_cache_udt",
        	mainKey: mainKey,  
        	secondaryKey: secondaryKey,
        	responseCallback: 	"window.callback_context_object.callback_method"
		});
  }
  
  
  
  function my_context_class_for_callback_context_falis2(value1)
  {
    //constructor function
  	this.key1 		= value1;  
  	this.set_key1 	= function(new_val) 
    {
    		this.key1 = new_val;
  	};
  	this.get_key1 	= function() 
    {
    	return this.key1;
  	};
    this.callback_method=function()
    {
      alert ("in callback. this.key1= " +this.key1);
    }
}

  function callback_context_falis2()
  {

    	var my_context_object = new my_context_class_for_callback_context_falis2("key1_value");
    
    
    	var mainKey 		= "1";

		});
  }
